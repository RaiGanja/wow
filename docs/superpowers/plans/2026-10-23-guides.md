# Гайды (этап 5 РЕДИЗАЙНА-2): план реализации

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Пять страниц гайдов и хаб `/wow/guides/` на трёх языках в стиле Icy Veins; гайды с главной переезжают туда, старые якоря перенаправляются.

**Architecture:** Своя сборка по образцу `build_tools.py`: шаблон `guide.src.html` (обвязка копируется с `tools.src.html`) + `content/guides.py` (тексты сразу на ru/en/uk) → `build_guides.py` рендерит блоки в HTML на Python и пишет `guides/<slug>/index.html` для трёх языков. Обвязка переводится через `translate()` и `i18n/guides.*.txt`, тело подставляется ПОСЛЕ перевода. Автолента новостей — GitHub Action пишет `assets/news.json`.

**Tech Stack:** Python 3 (стандартная библиотека), статический HTML/CSS/JS, GitHub Pages, GitHub Actions, Node + headless Chrome (`shot.mjs`) для скриншотов.

**Spec:** `docs/superpowers/specs/2026-10-23-guides-design.md` — читать вместе с планом.

- [ ] **Step 1: Написать проверку**

```python
"""Проверка собранного сайта: локальные ссылки существуют, нет @@, на EN-гайдах нет кириллицы.
Запуск: python tests/check_site.py"""
import os, re, sys
from urllib.parse import unquote, urlsplit

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
SKIP_DIRS = {'.git', '.github', '.claude', 'docs', 'content', 'i18n', 'tests', 'node_modules'}
ATTR = re.compile(r'(?:href|src)="([^"]+)"')
SRCSET = re.compile(r'srcset="([^"]+)"')
IDS = re.compile(r'\sid="([^"]+)"')


def pages():
    for d, dirs, files in os.walk(ROOT):
        dirs[:] = [x for x in dirs if x not in SKIP_DIRS]
        for f in files:
            if f == 'index.html':
                yield os.path.join(d, f)


def targets(html):
    out = ATTR.findall(html)
    for s in SRCSET.findall(html):
        out += [p.strip().split(' ')[0] for p in s.split(',') if p.strip()]
    return out


def main():
    problems, n = [], 0
    for page in pages():
        n += 1
        html = open(page, encoding='utf-8').read()
        rel = os.path.relpath(page, ROOT).replace(os.sep, '/')
        if '@@' in html:
            problems.append(f'{rel}: остался @@')
        if rel.startswith('en/guides/'):
            body = re.sub(r'<(script|style)\b.*?</\1>', '', html, flags=re.S)
            body = re.sub(r'<[^>]+>', ' ', body)
            cyr = re.findall(r'[А-Яа-яЁёІіЇїЄєҐґ][^<\n]{0,30}', body)
            if cyr:
                problems.append(f'{rel}: кириллица на EN: {cyr[:3]}')
        ids = set(IDS.findall(html))
        for t in targets(html):
            if t.startswith(('http:', 'https:', 'mailto:', 'data:', 'javascript:', '${')) or '${' in t:
                continue
            u = urlsplit(t)
            if not u.path:
                if u.fragment and u.fragment not in ids and not u.fragment.startswith(('pick=', 'team=')):
                    problems.append(f'{rel}: нет якоря #{u.fragment}')
                continue
            p = os.path.normpath(os.path.join(os.path.dirname(page), unquote(u.path)))
            if p.endswith(os.sep) or os.path.isdir(p):
                p = os.path.join(p, 'index.html')
            if not os.path.exists(p):
                problems.append(f'{rel}: нет файла {t}')
    for p in problems:
        print(p)
    if problems:
        print(f'FAIL {len(problems)} problems in {n} pages')
        sys.exit(1)
    print(f'OK {n} pages')


if __name__ == '__main__':
    main()
```

```python
            gaps += [f'{path}.{l}' for l in LANGS if not obj.get(l)]
        else:
            for k, v in obj.items():
                gaps += check_langs(v, f'{path}.{k}')
    elif isinstance(obj, (list, tuple)):
        for i, v in enumerate(obj):
            gaps += check_langs(v, f'{path}[{i}]')
    return gaps


def art_sets(art, root):
    ws = WIDTHS.get(art, (360, 600, 896))
    src = lambda ext: ', '.join(f'{root}assets/art/{art}-{w}.{ext} {w}w' for w in ws)
    return src('avif'), src('webp'), f'{root}assets/art/{art}-{ws[-1]}.webp'


def render_block(kind, data, lang, root):
    if kind == 'text':
        return ''.join(f'<p>{E(p, lang)}</p>' for p in data)
    if kind == 'steps':
        cards = ''.join(f'<li><span class="n">{i:02d}</span><b>{E(s["t"], lang)}</b><p>{E(s["d"], lang)}</p></li>'
                        for i, s in enumerate(data, 1))
        return f'<ol class="steps-c">{cards}</ol>'
    if kind == 'tiles':
        out = []
        for s in data:
            avif, webp, img = art_sets(s['art'], root)
            out.append(f'<a class="stile" href="{root}{s["href"]}"><picture><source type="image/avif" srcset="{avif}" sizes="300px">'
                       f'<img src="{img}" srcset="{webp}" sizes="300px" alt="" loading="lazy" decoding="async"></picture>'
                       f'<b>{E(s["t"], lang)}</b><span>{E(s["d"], lang)}</span></a>')
        return f'<div class="gtiles">{"".join(out)}</div>'
    if kind == 'table':
        head = [E(h, lang) for h in data['head']]
        th = ''.join(f'<th scope="col">{h}</th>' for h in head)
        rows = ''.join('<tr>' + ''.join(f'<td data-l="{h}">{E(c, lang)}</td>' for h, c in zip(head, r)) + '</tr>'
                       for r in data['rows'])
        return f'<table class="gtable"><thead><tr>{th}</tr></thead><tbody>{rows}</tbody></table>'
    if kind == 'tabs':
        btns = ''.join(f'<button type="button" class="pill" role="tab" data-tab="{x["id"]}" aria-selected="{"true" if i == 0 else "false"}">'
                       f'{E(x["t"], lang)}</button>' for i, x in enumerate(data))
        panels = ''.join(f'<div role="tabpanel" data-tab="{x["id"]}"{"" if i == 0 else " hidden"}>'
                         + ''.join(render_block(k, d, lang, root) for k, d in x['blocks']) + '</div>'
                         for i, x in enumerate(data))
        return f'<div class="gtabs"><div role="tablist">{btns}</div>{panels}</div>'
    if kind == 'faq':
        items = ''.join(f'<details id="{q["id"]}"><summary>{E(q["q"], lang)}</summary>'
                        + ''.join(f'<p>{E(a, lang)}</p>' for a in q['a']) + '</details>' for q in data)
        return f'<div class="gfaq">{items}</div>'
    if kind == 'tip':
        return (f'<div class="tipbox"><svg class="ico fi" aria-hidden="true"><use href="#i-fire"/></svg>'
                f'<p><span class="tl">{TIP[lang]}</span>{E(data, lang)}</p></div>')
    if kind == 'raw':
        return L(data, lang)
    raise ValueError(f'unknown block {kind!r}')


def render_guide(g, lang, root):
    toc = ''.join(f'<a href="#{s["id"]}">{E(s["title"], lang)}</a>' for s in g['sections'])
    body = '\n'.join(f'<section class="frame gsec" id="{s["id"]}"><h2>{E(s["title"], lang)}</h2>'
                     + ''.join(render_block(k, d, lang, root) for k, d in s['blocks']) + '</section>'
                     for s in g['sections'])
    return toc, body


def hub_body(lang, root):
    tiles = [{'t': g['title'], 'd': g['tile'], 'href': f'guides/{g["slug"]}/', 'art': g['art']} for g in GUIDES]
    return '', f'<section class="frame gsec" id="all">{render_block("tiles", tiles, lang, root)}</section>'


JS = {}   # slug -> JS живой части; наполняется в задачах 3, 5, 7, 8


def build_page(g, lang, tpl, base_css, sprite, src):
    sub = f'guides/{g["slug"]}/' if g['slug'] else 'guides/'
    here = page_dir(lang, sub)
    html = tpl.replace('@@BASE_CSS@@', base_css).replace('@@SPRITE@@', sprite)
    html, missing = translate(html, lang)                       # 1) обвязка
    root = rel(here, page_dir(lang))
    toc, body = render_guide(g, lang, root) if g['slug'] else hub_body(lang, root)
    others = [x for x in GUIDES if x['slug'] != g['slug']]
    more = render_block('tiles', [{'t': x['title'], 'd': x['tile'], 'href': f'guides/{x["slug"]}/', 'art': x['art']}
                                  for x in others], lang, root)
    avif, webp, img = art_sets(g['art'], '@@ROOT@@')
    crumb = (f'<a href="{root}guides/">{E(HUB["title"], lang)}</a><span aria-hidden="true">/</span><span>{E(g["title"], lang)}</span>'
             if g['slug'] else f'<span>{E(g["title"], lang)}</span>')
    js = g.get('js')
    rep = {
        'TITLE': E(g['title'], lang), 'LEAD': E(g['lead'], lang), 'CRUMB': crumb,
        'PILLS': ''.join(f'<span class="pill">{E(p, lang)}</span>' for p in g.get('pills', [])),
        'ART_AVIF': avif, 'ART_WEBP': webp, 'ART_IMG': img, 'TOC': toc, 'BODY': body, 'MORE': more,
        'SOURCES': ''.join(f'<li><a href="{u}" target="_blank" rel="noopener">{escape(n)}</a></li>' for n, u in g.get('sources', [])),
        'UPDATED': g.get('updated', ''),
        'JS': ((home_data(src) + '\n') if js in ('dungeons', 'professions') else '') + JS[js] if js else '',
    }
    for k, v in rep.items():                                     # 2) тело
        html = html.replace(f'@@{k}@@', v)
    html = html.replace('@@GUIDES@@', rel(here, page_dir(lang, 'guides/')))
    html = html.replace('@@HOME@@', root)
    for l in LANGS:
        html = html.replace(f'@@{l.upper()}@@', rel(here, page_dir(l, sub)))
        html = html.replace(f'@@CUR_{l.upper()}@@', 'true' if l == lang else 'false')
    html = html.replace('@@CANON@@', SITE + here)
    alt = '\n'.join(f'<link rel="alternate" hreflang="{l}" href="{SITE}{page_dir(l, sub)}">' for l in LANGS)
    alt += f'\n<link rel="alternate" hreflang="x-default" href="{SITE}{sub}">'
    html = html.replace('@@ALTERNATES@@', alt)
    html = localize(html, lang, here)
    if lang != 'ru':
        html = html.replace('assets/og.png', f'assets/og-{lang}.png')
    return here + 'index.html', html, missing


def main():
    gaps = check_langs(HUB) + [p for g in GUIDES for p in check_langs(g, g['slug'])] + check_langs(NEWS, 'NEWS')
    assert not gaps, gaps[:10]
    src = open(os.path.join(HERE, 'index.src.html'), encoding='utf-8').read()
    tpl = open(os.path.join(HERE, 'guide.src.html'), encoding='utf-8').read()
    base_css = src[src.index('<style>') + 7:src.index('</style>')].strip('\n')
    s0 = src.index('<svg width="0" height="0"')
    sprite = src[s0:src.index('</svg>', s0) + 6]

    pages, allmissing = [], []
    for lang in LANGS:
        for g in [HUB] + GUIDES:
            path, html, missing = build_page(g, lang, tpl, base_css, sprite, src)
            assert '@@' not in html, (path, re.findall(r'@@\w+@@', html)[:5])
            pages.append((path, html))
            allmissing += missing

    listing = '--list' in sys.argv
    if not listing:
        for path, html in pages:
            full = os.path.join(HERE, path)
            os.makedirs(os.path.dirname(full), exist_ok=True)
            open(full, 'w', encoding='utf-8', newline='\n').write(html)
    uniq = list(dict.fromkeys(allmissing))
    if listing:
        print('\n'.join(uniq))
    print(f'guides: {len(pages)}, missing: {len(uniq)}', file=sys.stderr)
    if uniq and not listing:
        sys.exit(1)


if __name__ == '__main__':
    main()
```

- [ ] **Step 2: `tools/fetch_news.py`**

```python
"""RSS Wowhead -> assets/news.json (только записи про Forever, 30 последних).
Сеть упала или RSS сломан -> ничего не пишем, выходим с кодом 0 (старый файл остаётся)."""
import json, os, sys, urllib.request
from email.utils import parsedate_to_datetime
from xml.etree import ElementTree as ET

FEED = 'https://www.wowhead.com/news/rss/all'
OUT = os.path.join(os.path.dirname(os.path.dirname(os.path.abspath(__file__))), 'assets', 'news.json')


def parse(xml):
    out = []
    for it in ET.fromstring(xml).iter('item'):
        title = (it.findtext('title') or '').strip()
        cats = ' '.join(c.text or '' for c in it.findall('category'))
        if 'forever' not in (title + ' ' + cats).lower():
            continue
        out.append({'date': parsedate_to_datetime(it.findtext('pubDate')).strftime('%Y-%m-%d'),
                    'title': title, 'url': (it.findtext('link') or '').strip()})
    return sorted(out, key=lambda x: x['date'], reverse=True)[:30]


def main():
    try:
        req = urllib.request.Request(FEED, headers={'User-Agent': 'gabella-wow-guide'})
        items = parse(urllib.request.urlopen(req, timeout=30).read().decode('utf-8'))
    except Exception as e:
        print(f'skip: {e}', file=sys.stderr)
        return
    old = json.load(open(OUT, encoding='utf-8')) if os.path.exists(OUT) else None
    if items and items != old:
        json.dump(items, open(OUT, 'w', encoding='utf-8', newline='\n'), ensure_ascii=False, indent=1)
        print(f'written {len(items)}')
    else:
        print('no change')


if __name__ == '__main__':
    main()
```
